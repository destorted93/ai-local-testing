# create a python script which create a complex random folders structure. It should start with fake_bin folder. Inside it start to create a random number of folders from 5 to 10 with random 10 characters long names. In each new folder, this task will repeat again. And so on, in a recursive way, for the next 5 levels.
import os
import string
import random

def create_random_folder_name(length=10):
    letters = string.ascii_lowercase + string.digits
    return ''.join(random.choice(letters) for _ in range(length))

def create_folders_structure(base_path, level=0, max_levels=5):
    if level > max_levels:
        return

    num_folders = random.randint(5, 10)
    for _ in range(num_folders):
        folder_name = create_random_folder_name()
        folder_path = os.path.join(base_path, folder_name)
        os.makedirs(folder_path, exist_ok=True)
        create_folders_structure(folder_path, level + 1, max_levels)
        # create also a random number of txt file fom 1 to 5 with random names
        num_files = random.randint(1, 5)
        for _ in range(num_files):
            file_name = create_random_folder_name() + '.txt'
            file_path = os.path.join(folder_path, file_name)
            open(file_path, 'w').close()

base_path = 'fake_bin'
create_folders_structure(base_path)
